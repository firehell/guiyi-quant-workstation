# RU2609只读价格复算

仅使用既有MDS、只读事务、权威Session及现有价格几何函数；没有新增策略计算器。输出路径必须不存在。以下是本轮实际执行的读取程序，保存到临时文件后运行，原程序输出与研究标注分开。

```python
import json,hashlib
from pathlib import Path
from datetime import date,datetime,UTC
from decimal import Decimal
from dotenv import load_dotenv
from lms405_replay import read_local
from lms405_sh_oct_case import summarize_scopes
from lms405_node_audit import geometry
load_dotenv('.env',override=False)
from app.db.session import SessionLocal
from app.db.readonly import readonly_transaction
from app.market_data.composition import build_market_data_service
from app.market_data.session_clock import resolved_session_windows_for_trading_day
from app.market_data.market_data_service import MarketDataError
r=dict(contract='RU2609',frequency='1m',window=['2026-08-10','2026-08-12'],observed_at=datetime.now(UTC).isoformat(),source_period='UNKNOWN',historical_known_at_verified=False,executable=False,fill=None,performance=None)
try:
 with SessionLocal() as s, readonly_transaction(s,timeout_seconds=120):
  m=build_market_data_service(s);x=read_local(m,'ru','RU2609','1m',date(2026,8,10),date(2026,8,12));bars=x.bars
  days=[date(2026,8,d) for d in (10,11,12)];exchange=m.catalog.exchange_for_symbol('ru')
  windows={d:resolved_session_windows_for_trading_day(s,exchange=exchange,symbol='ru',trading_day=d) for d in days}
  data=[dict(bar_end=b.bar_end.isoformat(),trading_day=b.trading_day.isoformat(),open=str(b.open),high=str(b.high),low=str(b.low),close=str(b.close)) for b in bars]
  complete=dict(status='WINDOW_READABLE',bar_count=len(bars),input_ohlc_sha256=hashlib.sha256(json.dumps(data,sort_keys=True).encode()).hexdigest(),coverage=[v.isoformat() for v in x.coverage],scopes=summarize_scopes(bars,windows,datetime.fromisoformat('2026-08-10T17:05:00+08:00')))
  complete['retrospective_aug10_plan_geometry']=[dict(lower=lo,upper=hi,source='A5330414436221055',**geometry(bars,datetime.fromisoformat('2026-08-10T17:14:00+08:00'),Decimal(lo),Decimal(hi))) for lo,hi in [('16705','16770'),('16855','16900'),('16955','16975'),('17020','17045')]]
 r.update(complete)
except MarketDataError as exc:r.update(status='BLOCKED',reason=exc.code)
except Exception as exc:r.update(status='BLOCKED',reason='LOCAL_READ_UNVERIFIED',error_type=type(exc).__name__)
import sys
with Path(sys.argv[1]).open('x') as f:json.dump(r,f,ensure_ascii=False,indent=2);f.write('\n')
print(json.dumps({k:r[k] for k in ['status','bar_count','reason'] if k in r}))
```
